import os
def start_service(service_name):
    os.system(f'systemctl start {service_name}')
    print(f'Service {service_name} started.')


def stop_service(service_name):
    os.system(f'systemctl stop {service_name}')
    print(f'Service {service_name} stopped.')

def restart_service(service_name):
    os.system(f'systemctl restart {service_name}')
    print(f'Service {service_name} restarted.')

def status_service(service_name):
    os.system(f'systemctl status {service_name}')   
    print(f'Status of service {service_name} displayed.')

if __name__ == "__main__":  
    
    service = input("Enter the service name to manage: ")

    start_service(service)
    status_service(service)
    restart_service(service)
    stop_service(service)

   def list_ec2_instances():
    ec2 = boto3.client('ec2')
    response = ec2.describe_instances()

    for reservation in response['Reservations']:
        for instance in reservation['Instances']:
            print(f'Instance ID: {instance["InstanceId"]}, State: {instance["State"]["Name"]}')
